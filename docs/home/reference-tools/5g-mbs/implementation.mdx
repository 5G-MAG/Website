import US_AUDIT from '@site/src/data/audit-5g-mbs-us.json';

{/* One section per part: only the features, components and baselines that part uses. */}
export const part = (keep) => {
  const features = AUDIT.features.filter(keep);
  const reqs = features.flatMap((f) => f.requirements);
  const pick = (o, keys) => Object.fromEntries(Object.entries(o).filter(([k]) => keys.has(k)));
  return {
    ...AUDIT,
    features,
    components: pick(AUDIT.components, new Set(reqs.map((r) => r.component))),
    releases: pick(AUDIT.releases, new Set(reqs.map((r) => r.release))),
    releaseNames: pick(AUDIT.releaseNames || {}, new Set(reqs.map((r) => r.release))),
  };
};
export const US = US_AUDIT;
export const COMMON = part((f) => f.mode === 'common');
export const BCAST = part((f) => f.mode === 'broadcast');
export const MCAST = part((f) => f.mode === 'multicast');
